    85→function statusBadge(s: string) {
    86→  const map: Record<string, { cls: string; label: string }> = {
    87→    HADIR: { cls: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300 border-emerald-200 dark:border-emerald-800', label: 'Hadir' },
    88→    IZIN: { cls: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300 border-amber-200 dark:border-amber-800', label: 'Izin' },
    89→    SAKIT: { cls: 'bg-sky-100 text-sky-700 dark:bg-sky-900/40 dark:text-sky-300 border-sky-200 dark:border-sky-800', label: 'Sakit' },
    90→    ALPHA: { cls: 'bg-rose-100 text-rose-700 dark:bg-rose-900/40 dark:text-rose-300 border-rose-200 dark:border-rose-800', label: 'Alpha' },
    91→  }
    92→  const m = map[s] ?? map.ALPHA
    93→  return (
    94→    <span className={`inline-flex items-center gap-1 px-2 py-0.5 rounded-md border text-xs font-semibold ${m.cls}`}>
    95→      {m.label}
    96→    </span>
    97→  )
    98→}

   100→function formatTime(d: string | null) {
   101→  if (!d) return '-'
   102→  return new Date(d).toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' })
   103→}
   104→
   105→function todayStr() {
   106→  const d = new Date()
   107→  const yyyy = d.getFullYear()
   108→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   109→  const dd = String(d.getDate()).padStart(2, '0')
   110→  return `${yyyy}-${mm}-${dd}`
   111→}
   112→
   113→function currentMonthStr() {
   114→  const d = new Date()
   115→  const yyyy = d.getFullYear()
   116→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   117→  return `${yyyy}-${mm}`
   118→}

   160→// ============ Shared kelompok loader hook ============
   161→function useKelompokList() {
   162→  const [list, setList] = useState<Kelompok[]>([])
   163→  const [loading, setLoading] = useState(true)
   164→  useEffect(() => {
   165→    (async () => {
   166→      try {
   167→        const res = await fetch('/api/kelompok')
   168→        if (!res.ok) throw new Error('Gagal')
   169→        const json = await res.json()
   170→        setList(json)
   171→      } catch {
   172→        toast.error('Gagal memuat daftar kelompok')
   173→      } finally {
   174→        setLoading(false)
   175→      }
   176→    })()
   177→  }, [])
   178→  return { list, loading }
   179→}

   196→  const [deleting, setDeleting] = useState(false)

   209→    } catch {
   210→      toast.error('Gagal memuat absensi harian')
   211→    } finally {
   212→      setLoading(false)
   213→    }
   214→  }, [tanggal, kelompokId, statusFilter])
   215→
   216→  useEffect(() => { fetchData() }, [fetchData])
   217→
   218→  // Stats
   219→  const stats = useMemo(() => {
   220→    const total = data.length
   221→    const hadir = data.filter((d) => d.status === 'HADIR').length
   222→    const izinSakit = data.filter((d) => d.status === 'IZIN' || d.status === 'SAKIT').length
   223→    const alpha = data.filter((d) => d.status === 'ALPHA').length

   232→

   236→    setSaving(true)
   237→    try {

   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify({ status: editStatus, keterangan: editKeterangan }),
   242→      })

   245→      toast.success('Status absensi diperbarui')
   246→      setEditTarget(null)
   247→      fetchData()
   248→    } catch (err: any) {
   249→      toast.error(err?.message || 'Gagal menyimpan')
   250→    } finally {
   251→      setSaving(false)
   252→    }
   253→  }
   254→

   256→    if (!deleteTarget) return

   259→      const res = await fetch(`/api/absensi/${deleteTarget.id}`, { method: 'DELETE' })

   262→      toast.success('Absensi berhasil dihapus')
   263→      setDeleteTarget(null)
   264→      fetchData()
   265→    } catch (err: any) {
   266→      toast.error(err?.message || 'Gagal menghapus')

   268→      setDeleting(false)
   269→    }
   270→  }

   708→function QrScannerTab() {

   717→  const [selectedMhsId, setSelectedMhsId] = useState('')
   718→  const [kelompokId, setKelompokId] = useState('')
   719→  const [status, setStatus] = useState('HADIR')

   721→
   722→  // QR preview state — encode a dummy mahasiswa ID
   723→  const [qrDataUrl, setQrDataUrl] = useState<string>('')
   724→  const dummyPayload = useMemo(() => {
   725→    // Build a small payload that contains a mahasiswa ID
   726→    return JSON.stringify({ type: 'KKN-PLP-ABSENSI', mhsId: selectedMhsId || 'SCAN-DEMO' })
   727→  }, [selectedMhsId])
   728→
   729→  useEffect(() => {
   730→    // Load mahasiswa list (lightweight, just nim+nama+id)
   731→    (async () => {
   732→      try {
   733→        const res = await fetch('/api/mahasiswa')
   734→        if (!res.ok) throw new Error('Gagal')
   735→        const json = await res.json()
   736→        setMhsList(json)
   737→      } catch {
   738→        // silent — toast error too noisy
   739→      }
   740→    })()
   741→  }, [])
   742→
   743→  useEffect(() => {
   744→    // Generate QR code as data URL
   745→    QRCode.toDataURL(dummyPayload, { width: 256, margin: 2, color: { dark: '#0f172a', light: '#ffffff' } })
   746→      .then(setQrDataUrl)
   747→      .catch(() => setQrDataUrl(''))
   748→  }, [dummyPayload])

   751→    setLoadingRecent(true)
   752→    try {
   753→      // get last 10 created (no filter)
   754→      const res = await fetch('/api/absensi?search=')
   755→      if (!res.ok) throw new Error('Gagal')
   756→      const json: Absensi[] = await res.json()
   757→      // Sort by createdAt desc and take 10
   758→      const sorted = [...json].sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()).slice(0, 10)
   759→      setRecentScans(sorted)
   760→    } catch {
   761→      // silent
   762→    } finally {
   763→      setLoadingRecent(false)
   764→    }
   765→  }, [])
   766→
   767→  useEffect(() => { fetchRecent() }, [fetchRecent])
   768→
   769→  // Resolve NIM scan input → mahasiswaId
   770→  const resolvedMhsId = useMemo(() => {
   771→    const trimmed = scanInput.trim()
   772→    if (!trimmed) return selectedMhsId
   773→    // If matches an id directly
   774→    const byId = mhsList.find((m) => m.id === trimmed)
   775→    if (byId) return byId.id
   776→    // By NIM
   777→    const byNim = mhsList.find((m) => m.nim.toLowerCase() === trimmed.toLowerCase())
   778→    if (byNim) return byNim.id
   779→    return selectedMhsId
   780→  }, [scanInput, selectedMhsId, mhsList])
   781→
   782→  const resolvedMhs = useMemo(() => mhsList.find((m) => m.id === resolvedMhsId) || null, [mhsList, resolvedMhsId])
   783→

   788→      return
   789→    }

   796→      const today = todayStr()

   798→        method: 'POST',
   799→        headers: { 'Content-Type': 'application/json' },
   800→        body: JSON.stringify({
   801→          mahasiswaId: resolvedMhsId,
   802→          kelompokId,
   803→          tanggal: today,
   804→          status,
   805→          keterangan: keterangan || undefined,
